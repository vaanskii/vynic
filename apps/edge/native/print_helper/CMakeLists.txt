cmake_minimum_required(VERSION 3.15)
cmake_policy(SET CMP0091 NEW)
project(vynic_print_helper LANGUAGES CXX)

set(_spooler "${CMAKE_CURRENT_LIST_DIR}/../../../operations/windows/runner")
add_library(vynic_print_helper_session STATIC protocol.cpp session.cpp)
target_compile_features(vynic_print_helper_session PUBLIC cxx_std_17)
target_include_directories(vynic_print_helper_session PUBLIC "${_spooler}")
if(MSVC)
  set_property(TARGET vynic_print_helper_session PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_compile_options(vynic_print_helper_session PRIVATE /W4 /WX /EHsc)
else()
  target_compile_options(vynic_print_helper_session PRIVATE -Wall -Wextra -Werror)
endif()

add_executable(print_helper_protocol_test protocol_test.cpp "${_spooler}/windows_spooler.cpp")
target_link_libraries(print_helper_protocol_test PRIVATE vynic_print_helper_session)
target_compile_features(print_helper_protocol_test PRIVATE cxx_std_17)
if(MSVC)
  set_property(TARGET print_helper_protocol_test PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_compile_options(print_helper_protocol_test PRIVATE /W4 /WX /EHsc)
else()
  target_compile_options(print_helper_protocol_test PRIVATE -Wall -Wextra -Werror)
endif()
enable_testing()
add_test(NAME print_helper_fake_protocol COMMAND print_helper_protocol_test)

if(WIN32)
  add_executable(vynic_print_helper main_windows.cpp "${_spooler}/windows_spooler.cpp" "${_spooler}/windows_spooler_native.cpp")
  target_compile_features(vynic_print_helper PRIVATE cxx_std_17)
  target_compile_definitions(vynic_print_helper PRIVATE NOMINMAX UNICODE _UNICODE)
  target_link_libraries(vynic_print_helper PRIVATE vynic_print_helper_session winspool)
  set_property(TARGET vynic_print_helper PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_compile_options(vynic_print_helper PRIVATE /W4 /WX /EHsc)
endif()
