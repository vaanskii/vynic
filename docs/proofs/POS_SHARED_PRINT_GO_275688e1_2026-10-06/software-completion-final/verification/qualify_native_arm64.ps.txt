function Cache-Snapshot {
  $paths = @(
    (Join-Path $cmakeBin 'cmake.exe'),
    (Join-Path $cmakeBin 'ctest.exe'),
    (Join-Path $vc 'lib\arm64\libcmt.lib'),
    (Join-Path $vc 'lib\arm64\libcpmt.lib'),
    (Join-Path $vc 'lib\arm64\libvcruntime.lib'),
    (Join-Path $vc 'lib\arm64\oldnames.lib'),
    (Join-Path $kit ('Lib\'+$sdkVersion+'\um\arm64\WinSpool.Lib')),
    (Join-Path $kit ('Lib\'+$sdkVersion+'\um\arm64\kernel32.Lib')),
    (Join-Path $kit ('Lib\'+$sdkVersion+'\ucrt\arm64\libucrt.lib'))
  )
  $directories = @((Join-Path $vc 'bin\Hostarm64\arm64'), (Join-Path $vc 'include'))
  foreach ($include in @('shared','um','ucrt')) { $directories += Join-Path $kit ('Include\'+$sdkVersion+'\'+$include) }
  foreach ($directory in $directories) {
    if (-not (Test-Path -LiteralPath $directory -PathType Container)) { throw "Cached tool/header directory missing: $directory" }
    $paths += @(Get-ChildItem -LiteralPath $directory -Recurse -File | Select-Object -ExpandProperty FullName)
  }
  $hashes = [ordered]@{}
  foreach ($path in @($paths | Sort-Object -Unique)) {
    if (-not (Test-Path -LiteralPath $path -PathType Leaf)) { throw "Cached ARM64 input missing: $path" }
    $hashes[$path] = (Get-FileHash -LiteralPath $path -Algorithm SHA256).Hash.ToLowerInvariant()
  }
  return $hashes
}

function Machine([string]$Path) {
  $bytes = [IO.File]::ReadAllBytes($Path)
  $offset = [BitConverter]::ToInt32($bytes, 60)
  if ($offset -lt 64 -or $offset + 6 -ge $bytes.Length -or [BitConverter]::ToUInt32($bytes, $offset) -ne 17744) { throw 'Invalid PE image' }
  return [BitConverter]::ToUInt16($bytes, $offset + 4)
}

function Run-Checked([string]$Label, [string]$Program, [string[]]$Arguments) {

  & $Program @Arguments *> $log
  $code = $LASTEXITCODE
  Get-Content -LiteralPath $log
  if ($code -ne 0) { throw "$Label failed with exit $code" }
}
