function Verify-Source([string]$root, $manifest) {
  $entries = @($manifest.files.PSObject.Properties)
  if ($entries.Count -ne 163 -or $manifest.count -ne 163) { throw 'Exact 163-file manifest required' }
  $disk = @(Get-ChildItem -LiteralPath $root -Recurse -File -Force)
  if ($disk.Count -ne 163) { throw 'Extracted source count differs from exact frozen snapshot' }
  foreach ($entry in $entries) {
    $p = Join-Path $root ($entry.Name.Replace('/', '\'))
    $item = Get-Item -LiteralPath $p -Force
    if (($item.Attributes -band [IO.FileAttributes]::ReparsePoint) -ne 0 -or $item.Length -ne $entry.Value.bytes -or (Get-FileHash -LiteralPath $p -Algorithm SHA256).Hash.ToLowerInvariant() -ne $entry.Value.sha256) { throw ('Source mismatch: ' + $entry.Name) }
  }
}
